# -*- coding: utf-8 -*-
"""v46 (17-sep-2026, decisión de FR): en los tramos del Anexo A sin dato de carriles, la red básica y la red
secundaria (libre) asumen 1 carril por sentido (factor de distribución 1.00) en lugar del 2 por omisión;
las demás redes siguen con 2. /api/tramo devuelve además `carriles_fuente` y `red`. Idempotente.
Anclas por expresión regular (v46b) para no depender del orden exacto de columnas del SELECT.
Uso: python3 parche_v46_carriles.py [ruta a main.py]"""
import io, sys, os, re
p = sys.argv[1] if len(sys.argv) > 1 else os.path.expanduser("~/hga-api/main.py")
s = io.open(p, encoding="utf-8").read()
orig = s

# 1) helper carriles_de(cp) -> carriles_de(cp, red)
A = '''def carriles_de(cp):
    """Carriles por sentido del Anexo A 2025 (tabla `carriles`); None si no hay dato."""
    try:
        r = DATOS["db"].execute("SELECT carriles FROM carriles WHERE cp=?", (cp,)).fetchone()
        return int(r[0]) if r and r[0] else None
    except Exception:
        return None
'''
B = '''REDES_UN_CARRIL = ("BASICA", "SECUNDARIA")   # red federal libre: 1 carril por sentido cuando el Anexo A no trae dato


def carriles_de(cp, red=None):
    """Carriles por sentido del Anexo A 2025 (tabla `carriles`) y su origen. Sin dato: 1 en red básica y
    secundaria (FR, 17-sep-2026), None (el cliente aplica 2) en las demás redes."""
    try:
        r = DATOS["db"].execute("SELECT carriles FROM carriles WHERE cp=?", (cp,)).fetchone()
        if r and r[0]:
            return int(r[0]), "Anexo A 2025"
    except Exception:
        pass
    if (red or "").upper() in REDES_UN_CARRIL:
        return 1, "red básica/secundaria sin dato en el Anexo A: 1 por omisión"
    return None, None
'''
if "def carriles_de(cp, red=None)" not in s:
    assert s.count(A) == 1, "ancla carriles_de (helper no encontrado o ya distinto)"
    s = s.replace(A, B)

# 2) SELECT del tramo: agregar la columna `red` si no está
def _sel(m):
    cols = m.group(1)
    if "estados" not in cols or re.search(r"\bred\b", cols):   # solo el SELECT de /api/tramo (trae `estados`)
        return m.group(0)
    return "SELECT " + cols + ",red FROM tramos WHERE id=?"
s, n_sel = re.subn(r"SELECT ([A-Za-z_,\s]+?) FROM tramos WHERE id=\?", _sel, s)
assert re.search(r"SELECT [A-Za-z_,\s]*estados[A-Za-z_,\s]*\bred\b[A-Za-z_,\s]* FROM tramos WHERE id=\?", s), \
    "ancla select tramo (no se encontró el SELECT de /api/tramo con `estados`)"

# 3) respuesta de /api/tramo
E = '"carriles": carriles_de(f["cp"]),'
F = ('"carriles": carriles_de(f["cp"], f["red"])[0],\n'
     '            "carriles_fuente": carriles_de(f["cp"], f["red"])[1],\n'
     '            "red": f["red"],')
if "carriles_fuente" not in s:
    assert s.count(E) == 1, "ancla respuesta tramo ('\"carriles\": carriles_de(f[\"cp\"]),' no encontrada)"
    s = s.replace(E, F)

if s != orig:
    io.open(p, "w", encoding="utf-8").write(s)
    print("PARCHE_V46_OK", p)
else:
    print("PARCHE_V46_YA_APLICADO", p)
