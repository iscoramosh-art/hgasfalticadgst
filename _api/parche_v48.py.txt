def sub1(s, patron, repl, nombre, flags=0, ya=None):
    if ya and ya in s:
        return s
    n = len(re.findall(patron, s, flags))
    if n != 1:
        fallas.append("%s: %d coincidencias" % (nombre, n))
        return s
    return re.sub(patron, repl, s, count=1, flags=flags)
