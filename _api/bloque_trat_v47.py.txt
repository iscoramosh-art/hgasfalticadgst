def emul_opts(tipo, tdpa):
    poli = tdpa > TDPA_POLIMERO
    if tipo == "micro":
        return [["ECCR-62P", "ECCR-62P, rompimiento rápido controlado (modificada con polímero)"],
                ["ECCL-62P", "ECCL-62P, rompimiento lento controlado (modificada con polímero)"]]
    if tipo == "premez":
        if poli:
            return [["ECL-62P", "ECL-62P, modificada con polímero para mezclas en frío en planta (obligatoria para TDPA > 5,000)"]]
        return [["ECM-65", "ECM-65, rompimiento medio (mezclado en planta)"],
                ["ECL-62P", "ECL-62P, modificada con polímero para mezclas en frío en planta"]]
    if poli:
        return [["ECR-68P", "ECR-68P, modificada con polímero (obligatoria para TDPA > 5,000)"]]
    return [["ECR-65", "ECR-65, rompimiento rápido"],
            ["ECR-68", "ECR-68, rompimiento rápido (clima cálido)"],
            ["ECR-68P", "ECR-68P, modificada con polímero"]]


def tratamientos_admitidos(tdpa, agriet="lt10"):
    """[(tipo, etiqueta, recomendado)] según TDPA y agrietamiento (FR, 21-sep-2026). Vacío = no procede."""
    if agriet == "gt50" or tdpa > TDPA_MAX_TRAT:
        return []
    if agriet in ("10-25", "25-50"):
        return [("scrub", LABEL_TRAT["scrub"], True)]
    if tdpa <= TDPA_RIEGO:
        return [("riego", LABEL_TRAT["riego"], True), ("premez", LABEL_TRAT["premez"], False),
                ("micro", LABEL_TRAT["micro"], False)]
    return [("premez", LABEL_TRAT["premez"], True), ("micro", LABEL_TRAT["micro"], False)]


def recomendar_trat(tdpa, ga, agriet="lt10"):
    calido = (ga if isinstance(ga, int) else 64) >= 70
    poli = tdpa > TDPA_POLIMERO
    adm = tratamientos_admitidos(tdpa, agriet)
    tipo = next((t for t, _, r in adm if r), adm[0][0] if adm else "riego")
    emul_riego = "ECR-68P" if poli else ("ECR-68" if calido else "ECR-65")
    emul_premez = "ECL-62P" if poli else "ECM-65"
    emul_micro = "ECCL-62P" if calido else "ECCR-62P"
    emul_scrub = "ECR-68P"
    mat_micro = "4A" if tdpa > 15000 else "4B"
    emul = {"riego": emul_riego, "premez": emul_premez, "micro": emul_micro, "scrub": emul_scrub}
    mat = {"riego": "3A", "premez": "3A", "micro": mat_micro, "scrub": "3A"}
    tdpa_txt = format(int(tdpa), ",d")
    if agriet in ("10-25", "25-50"):
        motivo = ("Agrietamiento %s: única opción el calafateo inducido (scrub seal), %s"
                  % (AGRIET_TXT[agriet].lower(),
                     "intervención óptima" if agriet == "10-25" else "solución paliativa de 1 a 3 años"))
    elif tdpa <= TDPA_RIEGO:
        motivo = "TDPA %s (hasta 10,000): se admiten riego de sello, premezclado y microaglomerado; se recomienda el riego de sello" % tdpa_txt
    else:
        motivo = "TDPA %s (de 10,000 a 50,000): se admiten premezclado y microaglomerado; se recomienda el riego de sello premezclado en planta" % tdpa_txt
    txt = (LABEL_TRAT[tipo] + ", " + emul[tipo] + ", clima " + ("cálido" if calido else "templado") +
           " (PG alto " + str(ga) + "). " + motivo +
           (". TDPA > 5,000: emulsión modificada con polímero obligatoria." if poli else "."))
    return {"tipo": tipo, "emul": emul[tipo], "mat": mat[tipo], "emul_riego": emul_riego, "emul_micro": emul_micro,
            "emul_premez": emul_premez, "emul_scrub": emul_scrub, "emul_por_tipo": emul, "mat_por_tipo": mat,
            "mat_riego": "3A", "mat_micro": mat_micro, "admitidos": [list(a) for a in adm], "texto": txt}

    return LABEL_TRAT.get(tipo, LABEL_TRAT["riego"])
